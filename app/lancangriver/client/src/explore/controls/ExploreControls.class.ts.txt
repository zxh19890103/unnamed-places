import { LatLng, latlngToSphere, sphereToLatlng } from '@/_3dtiles';
import { EARTH_RADIUS } from '@/calc/constants';
import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/Addons.js';

interface ExploreControlsEventMap {
  /**
   * every time the camera's view change ends.
   */
  end: { latlng: LatLng };
  /**
   * every time the camera's view changes
   */
  change: { latlng: LatLng };
  /**
   * when user click the earth, null if fail hit
   */
  click: { latlng: LatLng };
  /**
   * zoom change
   */
  zoom: {};
  state: { data: LiveState };
}

type Options = {};

type LiveState = {
  /**
   * distance from camera to target
   */
  distance: number;

  alt: number;
  latlng: LatLng;

  mode: 'orbit' | 'map';
};

export type ExploreControlsLiveState = LiveState;

OrbitControls;

export class ExploreControls extends THREE.Controls<ExploreControlsEventMap> {
  target: THREE.Vector3 = new THREE.Vector3();

  liveState: LiveState = {
    distance: 0,
    mode: 'orbit',
    alt: 0,
    latlng: { lat: 0, lng: 0 },
  };

  minDistance: number = 5_000;
  maxDistance: number = EARTH_RADIUS * 10;

  rotateSpeed: number = 1;
  zoomSpeed: number = 1;

  enableDamping: boolean = true;

  private _dispose: VoidFunction;
  private _disable: VoidFunction;
  private _checklow: VoidFunction;

  constructor(camera: THREE.PerspectiveCamera, domElement: HTMLElement, options: Options = {}) {
    super(camera, domElement);
    this._disable = this._enable();
    this._init();
  }

  private setLiveState(update: Partial<LiveState> = {}) {
    const pos = this.object.position;

    const distance = pos.distanceTo(this.target);

    const alt = pos.length() - EARTH_RADIUS;
    const latlng = sphereToLatlng(pos.x, pos.y, pos.z);

    this.liveState = {
      ...this.liveState,
      ...update,
      alt,
      distance,
      latlng,
    };

    this.dispatchEvent({
      type: 'state',
      data: this.liveState,
    });
  }

  private _init() {
    // check object's position
    const pos = this.object.position;

    let len = pos.length();

    const min = this.minDistance + EARTH_RADIUS_METERS;
    const max = this.maxDistance + EARTH_RADIUS_METERS;

    len = Math.max(min, Math.min(max, len));

    pos.setLength(len);
  }

  private _enable() {
    const dom = this.domElement as HTMLCanvasElement;

    let islow = false;

    const lowthrehold = 3_000; // 3 km

    const checklow = () => {
      if (islow) {
        const distance = this.object.position.distanceTo(this.target);

        if (distance > lowthrehold) {
          islow = false;
          this.setLiveState({ mode: 'orbit' });
        }
      } else {
        const distance = this.object.position.distanceTo(this.target) - EARTH_RADIUS;

        if (distance < lowthrehold) {
          islow = true;
          this.setLiveState({ mode: 'map' });
        }
      }
    };

    this._checklow = checklow;

    const onwheel_ORBIT = (event: WheelEvent) => {
      event.preventDefault();

      const dx = event.deltaX;
      const dy = event.deltaY;

      const isScroll = Number.isInteger(dx) && Number.isInteger(dy);
      const isPinch = !isScroll;

      const isShiftKey = event.metaKey || event.shiftKey;

      const position = new THREE.Vector3();
      const objectAlt = this.object.position.distanceTo(this.target) - EARTH_RADIUS;

      if (isScroll) {
        if (isShiftKey) {
          const deltaMeters = deriveMetersByZoomingDeltaPixel(dy, objectAlt);
          const currentMeters = this.object.position.distanceTo(this.target);

          const nextMeters = clampDistance(
            currentMeters,
            deltaMeters,
            this.minDistance,
            this.maxDistance,
          );

          position.subVectors(this.object.position, this.target);
          position.setLength(nextMeters);

          this.object.position.copy(this.target.clone().add(position));
          this.setLiveState();
          checklow();
        } else {
          const sensitivity = deriveSensitivityFromRotationDelta(dx, dy, objectAlt);
          const deltaX = -dx * sensitivity;
          const deltaY = dy * sensitivity;

          const offset = this.object.position.clone().sub(this.target);
          const spherical = new THREE.Spherical().setFromVector3(offset);

          spherical.theta -= deltaX;
          spherical.phi = THREE.MathUtils.clamp(spherical.phi + deltaY, 0.01, Math.PI - 0.01);

          const nextOffset = new THREE.Vector3().setFromSpherical(spherical);
          const nextDistance = nextOffset.length();
          const clampedDistance = clampDistance(
            nextDistance,
            0,
            this.minDistance,
            this.maxDistance,
          );
          nextOffset.setLength(clampedDistance);

          this.object.position.copy(nextOffset.add(this.target));

          this.object.lookAt(this.target);
          this.setLiveState();
        }
      } else {
        //
      }
    };

    /**
     * on low alt, it's very different
     * */
    const onwheel_MAP = (event: WheelEvent) => {
      event.preventDefault();

      const dx = event.deltaX;
      const dy = event.deltaY;

      const isScroll = Number.isInteger(dx) && Number.isInteger(dy);
      const isPinch = !isScroll;

      const isShiftKey = event.metaKey || event.shiftKey;

      if (isScroll) {
        if (isShiftKey) {
          const position = new THREE.Vector3();
          const deltaMeters = dy * 100;

          const currentMeters = this.object.position.distanceTo(this.target);
          const nextMeters = clampDistance(
            currentMeters,
            deltaMeters,
            this.minDistance,
            this.maxDistance,
          );

          position.subVectors(this.object.position, this.target);
          position.setLength(nextMeters);

          this.object.position.copy(this.target.clone().add(position));
          this.setLiveState();
          checklow();
        } else {
          const cameraOffset = this.object.position.clone().sub(this.target);

          const up = this.target.clone().normalize();
          const worldNorth = new THREE.Vector3(0, 1, 0);
          let east = worldNorth.clone().cross(up);

          if (east.lengthSq() < 1e-10) {
            east = new THREE.Vector3(1, 0, 0).cross(up);
          }
          east.normalize();

          const north = up.clone().cross(east).normalize();

          const nextTarget = this.target
            .clone()
            .addScaledVector(east, dx)
            .addScaledVector(north, -dy)
            .normalize()
            .multiplyScalar(EARTH_RADIUS);

          this.target.copy(nextTarget);
          this.object.position.copy(nextTarget.clone().add(cameraOffset));
          this.object.lookAt(this.target);
          this.setLiveState();
        }
      } else {
        //
      }
    };

    const onwheel = (event: WheelEvent) => {
      if (islow) {
        onwheel_MAP(event);
      } else {
        onwheel_ORBIT(event);
      }
    };

    dom.addEventListener('wheel', onwheel);

    this._disable = () => {
      console.log('disposed');
      dom.removeEventListener('wheel', onwheel);
    };

    return this._disable;
  }

  setObjectAt(latlng: LatLng, alt: number = 100_000) {
    const position = latlngToSphere(latlng.lat, latlng.lng, alt);
    this.object.position.copy(position);
    this._checklow();
    this.object.lookAt(this.target);
  }

  setObjectAtLowAlt(latlng: LatLng, alt: number = 1000) {
    const target = latlngToSphere(latlng.lat, latlng.lng, 0);
    const position = latlngToSphere(latlng.lat, latlng.lng, alt);

    this.target.copy(target);
    this.object.position.copy(position);
    this._checklow();
    this.object.lookAt(this.target);
  }

  connect(_element?: HTMLElement | SVGElement): void {
    // no-op
    console.log('connect');
  }

  disconnect(): void {
    // no-op
  }

  dispose(): void {
    // no-op
    this._disable?.();
    this._dispose?.();
  }

  update(_delta?: number): void {
    // no-op
  }
}

const distanceScale = 1_000;
const EARTH_RADIUS_METERS = 6_371_000;

function clampDistance(
  currentDistance: number,
  delta: number,
  minDistance: number,
  maxDistance: number,
) {
  const min = minDistance + EARTH_RADIUS_METERS;
  const max = maxDistance + EARTH_RADIUS_METERS;

  let _currDist = currentDistance;

  if (_currDist > max) {
    _currDist = max;
  }

  if (_currDist < min) {
    _currDist = min;
  }

  if (delta < 0 && _currDist + delta < min) {
    return min;
  } else if (delta > 0 && _currDist + delta > max) {
    return max;
  } else {
    return _currDist + delta;
  }
}

export const deriveSensitivityFromRotationDelta = (dx: number, dy: number, far: number) => {
  const distanceToSurface = Math.max(far, 0);
  const log2 = 1 + Math.log2(1 + distanceToSurface / distanceScale);
  const factor = 0.000001 * log2;
  return Math.max(Math.abs(dx), Math.abs(dy)) * factor;
};

export const deriveMetersByZoomingDeltaPixel = (delta: number, far: number) => {
  const distanceToSurface = Math.max(far, 0);
  if (distanceToSurface < 10_000 && delta < 0) {
    return delta * 0.0008 * Math.log2(distanceToSurface);
  } else {
    const log2 = 1 + Math.log2(1 + distanceToSurface / distanceScale);
    const factor = 1000 * log2;
    return delta * factor;
  }
};
